static ProjectionSummary Project(double[] vector, int edgeCount, int dimG,
    double[][] eBasis, double[][] wBasis, double[][] cBasis)
{
    double total = Dot(vector, vector);
    double eSquared = ProjectedNormSquared(vector, edgeCount, dimG, eBasis);
    double wSquared = ProjectedNormSquared(vector, edgeCount, dimG, wBasis);
    double cSquared = ProjectedNormSquared(vector, edgeCount, dimG, cBasis);
    double cPerpSquared = System.Math.Max(0.0, total - cSquared);
    var gram = new double[3, 3];
    foreach (double[] basis in cBasis)
    {
        var coefficient = new double[3];
        for (int a = 0; a < dimG; a++)
            for (int edge = 0; edge < edgeCount; edge++)
                coefficient[a] += basis[edge] * vector[edge * dimG + a];
        for (int a = 0; a < dimG; a++)
            for (int b = 0; b < dimG; b++) gram[a, b] += coefficient[a] * coefficient[b];
    }
    double[] eigenvalues = Symmetric3Eigenvalues(gram);
    double trace = eigenvalues.Sum();
    double alignment = trace > 0.0 ? eigenvalues[2] / trace : 0.0;
    double valleyDistance = cPerpSquared + eigenvalues[0] + eigenvalues[1];
    double scale = System.Math.Max(1.0, total);
    double residual = System.Math.Max(System.Math.Abs(cSquared - eSquared - wSquared),
        System.Math.Abs(total - cSquared - cPerpSquared)) / scale;
    return new ProjectionSummary(total, eSquared, wSquared, cSquared, cPerpSquared,
        eigenvalues, alignment, valleyDistance, residual);
}

static double ProjectedNormSquared(double[] vector, int edgeCount, int dimG, double[][] basis)
{
    double sum = 0.0;
    foreach (double[] direction in basis)
        for (int a = 0; a < dimG; a++)
        {
            double coefficient = 0.0;
            for (int edge = 0; edge < edgeCount; edge++) coefficient += direction[edge] * vector[edge * dimG + a];
            sum += coefficient * coefficient;
        }
    return sum;
}

static double[] Symmetric3Eigenvalues(double[,] input)
{
    var a = (double[,])input.Clone();
    for (int sweep = 0; sweep < 32; sweep++)
    {
        (int p, int q) = System.Math.Abs(a[0, 1]) >= System.Math.Abs(a[0, 2])
            ? (System.Math.Abs(a[0, 1]) >= System.Math.Abs(a[1, 2]) ? (0, 1) : (1, 2))
            : (System.Math.Abs(a[0, 2]) >= System.Math.Abs(a[1, 2]) ? (0, 2) : (1, 2));
        if (System.Math.Abs(a[p, q]) <= 1e-15 * System.Math.Max(1.0,
            System.Math.Max(System.Math.Abs(a[p, p]), System.Math.Abs(a[q, q])))) break;
        double angle = 0.5 * System.Math.Atan2(2.0 * a[p, q], a[q, q] - a[p, p]);
        double c = System.Math.Cos(angle), s = System.Math.Sin(angle);
        double app = c * c * a[p, p] - 2.0 * s * c * a[p, q] + s * s * a[q, q];
        double aqq = s * s * a[p, p] + 2.0 * s * c * a[p, q] + c * c * a[q, q];
        for (int r = 0; r < 3; r++)
        {
            if (r == p || r == q) continue;
            double arp = c * a[r, p] - s * a[r, q];
            double arq = s * a[r, p] + c * a[r, q];
            a[r, p] = a[p, r] = arp;
            a[r, q] = a[q, r] = arq;
        }
        a[p, p] = app; a[q, q] = aqq; a[p, q] = a[q, p] = 0.0;
    }
    double[] result = [System.Math.Max(0.0, a[0, 0]), System.Math.Max(0.0, a[1, 1]), System.Math.Max(0.0, a[2, 2])];
    Array.Sort(result);
    return result;
}

static bool IsClosed(double[] generator, SimplicialMesh mesh)
{
    for (int face = 0; face < mesh.FaceCount; face++)
    {
        double sum = 0.0;
        int[] edges = mesh.FaceBoundaryEdges[face];
        int[] orientations = mesh.FaceBoundaryOrientations[face];
        for (int i = 0; i < edges.Length; i++) sum += orientations[i] * generator[edges[i]];
        if (sum != 0.0) return false;
    }
    return true;
}

static void AddOrthonormal(List<double[]> basis, double[] candidate, double threshold)
{
    ProjectOut(candidate, basis);
    double norm = System.Math.Sqrt(Dot(candidate, candidate));
    if (norm <= threshold) return;
    for (int i = 0; i < candidate.Length; i++) candidate[i] /= norm;
    basis.Add(candidate);
}

static void ProjectOut(double[] candidate, IEnumerable<double[]> basis)
{
    for (int pass = 0; pass < 2; pass++)
        foreach (double[] direction in basis)
        {
            double coefficient = Dot(candidate, direction);
            for (int i = 0; i < candidate.Length; i++) candidate[i] -= coefficient * direction[i];
        }
}

static double MaximumBasisDefect(double[][] basis)
{
    double maximum = 0.0;
    for (int i = 0; i < basis.Length; i++)
        for (int j = 0; j <= i; j++)
            maximum = System.Math.Max(maximum, System.Math.Abs(Dot(basis[i], basis[j]) - (i == j ? 1.0 : 0.0)));
    return maximum;
}

static double[] Difference(double[] a, double[] b)
{
    var result = new double[a.Length];
    for (int i = 0; i < result.Length; i++) result[i] = a[i] - b[i];
    return result;
}

static double RelativeVectorError(double[] actual, double[] expected)
{
    double numerator = 0.0, denominator = 0.0;
    for (int i = 0; i < actual.Length; i++)
    {
        double delta = actual[i] - expected[i];
        numerator += delta * delta;
        denominator += expected[i] * expected[i];
    }
    return System.Math.Sqrt(numerator / System.Math.Max(1.0, denominator));
}

static bool BitIdentical(double[] a, double[] b) => a.Length == b.Length
    && a.Zip(b).All(x => BitConverter.DoubleToInt64Bits(x.First) == BitConverter.DoubleToInt64Bits(x.Second));
static double Dot(double[] a, double[] b)
{
    double sum = 0.0;
    for (int i = 0; i < a.Length; i++) sum += a[i] * b[i];
    return sum;
}
static double? Reportable(double value) => double.IsFinite(value) ? value : null;
static string Sha(string path) => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(path))).ToLowerInvariant();
static string VectorHash(double[] values)
{
    byte[] bytes = new byte[values.Length * sizeof(double)];
    Buffer.BlockCopy(values, 0, bytes, 0, bytes.Length);
    return Convert.ToHexString(SHA256.HashData(bytes)).ToLowerInvariant();
}

static double Uniform(Xoshiro rng) => ((rng.Next() >> 11) + 0.5) / 9007199254740992.0;
static double Gauss(Xoshiro rng)
{
    double u1 = Uniform(rng), u2 = Uniform(rng);
    return System.Math.Sqrt(-2.0 * System.Math.Log(u1)) * System.Math.Cos(2.0 * System.Math.PI * u2);
}
static ulong[] ExpandSeed(ulong seed)
{
    ulong state = seed;
    ulong Next()
    {
        state += 0x9E3779B97F4A7C15UL;
        ulong z = state;
        z = (z ^ (z >> 30)) * 0xBF58476D1CE4E5B9UL;
        z = (z ^ (z >> 27)) * 0x94D049BB133111EBUL;
        return z ^ (z >> 31);
    }
    return [Next(), Next(), Next(), Next()];
}

sealed class Xoshiro(ulong[] state)
{
    private ulong _s0 = state[0], _s1 = state[1], _s2 = state[2], _s3 = state[3];
    public ulong Next()
    {
        ulong result = RotateLeft(_s1 * 5, 7) * 9;
        ulong t = _s1 << 17;
        _s2 ^= _s0; _s3 ^= _s1; _s1 ^= _s2; _s0 ^= _s3; _s2 ^= t; _s3 = RotateLeft(_s3, 45);
        return result;
    }
    private static ulong RotateLeft(ulong x, int k) => (x << k) | (x >> (64 - k));
}
