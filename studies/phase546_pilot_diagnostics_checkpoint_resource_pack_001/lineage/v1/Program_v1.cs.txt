static ResourceAssessment Assess(ResourceRequest request, JsonElement rules)
{
    string requiredTopology = rules.GetProperty("requiredTopologyId").GetString()!;
    int requiredDimensions = rules.GetProperty("requiredDimensions").GetInt32();
    int maximumChains = rules.GetProperty("maximumChains").GetInt32();
    try
    {
        long cpu = checked((long)request.Chains * request.Trajectories * request.LeapfrogSteps
            * request.DegreesOfFreedom * rules.GetProperty("cpuTicksPerForceDegreeOfFreedom").GetInt64());
        long memory = checked((long)request.Chains * (rules.GetProperty("fixedBytesPerChain").GetInt64()
            + checked((long)request.DegreesOfFreedom * rules.GetProperty("bytesPerDegreeOfFreedom").GetInt64())));
        string? refusal = request.TopologyId != requiredTopology ? "topology-mismatch"
            : request.Dimensions != requiredDimensions ? "dimension-mismatch"
            : request.Extent <= 0 || request.Chains <= 0 || request.Chains > maximumChains
                || request.DegreesOfFreedom != checked(45 * request.Extent * request.Extent * request.Extent * request.Extent)
                || request.Trajectories <= 0 || request.LeapfrogSteps <= 0 ? "invalid-shape"
            : cpu >= rules.GetProperty("maximumAggregateCpuTicks").GetInt64() ? "cpu-boundary-or-limit-exceeded"
            : memory >= rules.GetProperty("maximumPeakBytes").GetInt64() ? "memory-boundary-or-limit-exceeded"
            : null;
        return new ResourceAssessment(refusal is null, refusal, cpu, memory);
    }
    catch (OverflowException)
    {
        return new ResourceAssessment(false, "checked-arithmetic-overflow", long.MaxValue, long.MaxValue);
    }
}

static bool ThrowsInvalidData(Action action)
{
    try { action(); return false; }
    catch (InvalidDataException) { return true; }
}

static byte[] NonCanonicalPayload(CheckpointState state)
{
    using var stream = new MemoryStream();
    using (var writer = new Utf8JsonWriter(stream))
    {
        writer.WriteStartObject();
        writer.WriteString("topologyId", state.TopologyId);
        writer.WriteString("formatId", state.FormatId);
        writer.WriteString("actionFingerprint", state.ActionFingerprint);
        writer.WriteString("packFingerprint", state.PackFingerprint);
        writer.WriteNumber("extent", state.Extent);
        writer.WriteNumber("dimensions", state.Dimensions);
        writer.WriteNumber("degreesOfFreedom", state.DegreesOfFreedom);
        writer.WriteNumber("beta", state.Beta);
        writer.WriteString("thetaRule", state.ThetaRule);
        writer.WriteString("chainId", state.ChainId);
        writer.WriteNumber("seed", state.Seed);
        writer.WriteNumber("trajectoryIndex", state.TrajectoryIndex);
        writer.WriteNumber("warmupCompleted", state.WarmupCompleted);
        writer.WriteNumber("retainedCompleted", state.RetainedCompleted);
        writer.WriteNumber("stepSize", state.StepSize);
        writer.WriteNumber("leapfrogSteps", state.LeapfrogSteps);
        writer.WriteString("rngAlgorithm", state.RngAlgorithm);
        writer.WritePropertyName("rngStateWords");
        JsonSerializer.Serialize(writer, state.RngStateWords);
        writer.WritePropertyName("position");
        JsonSerializer.Serialize(writer, state.Position);
        writer.WriteEndObject();
    }
    return stream.ToArray();
}

static byte[] Envelope(byte[] payload)
{
    using var stream = new MemoryStream();
    using (var writer = new Utf8JsonWriter(stream))
    {
        writer.WriteStartObject();
        writer.WriteString("checksumAlgorithm", "SHA-256");
        writer.WriteString("payloadSha256", Convert.ToHexString(SHA256.HashData(payload)).ToLowerInvariant());
        writer.WritePropertyName("payload");
        writer.WriteRawValue(payload);
        writer.WriteEndObject();
    }
    return stream.ToArray();
}

static string Sha(string path) => Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(path))).ToLowerInvariant();
