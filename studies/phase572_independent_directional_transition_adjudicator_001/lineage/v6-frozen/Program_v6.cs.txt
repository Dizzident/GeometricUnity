Evaluation Evaluate(double[] position)
{

    return new Evaluation(joint.Objective, joint.GradOmega);

static string ResolveNonfiniteTerminal(string[] taxonomy, string existing, string[] paths)
{
    int existingIndex = Array.IndexOf(taxonomy, existing);
    if (existingIndex < 0) existingIndex = 0;
    int nonfiniteIndex = taxonomy.Length - 1;
    foreach (string path in paths)
    {
        int pathIndex = path.StartsWith("$.independentBattery", StringComparison.Ordinal) ? 2
            : path.StartsWith("$.committedReplay", StringComparison.Ordinal) ? 4
            : path.StartsWith("$.directionalAdjudication", StringComparison.Ordinal)
                || path.StartsWith("$.projectors", StringComparison.Ordinal) ? 5
            : path.StartsWith("$.transitionAdjudication", StringComparison.Ordinal) ? 6
            : 0;
        nonfiniteIndex = System.Math.Min(nonfiniteIndex, pathIndex);
    }
    return paths.Length == 0 ? taxonomy[existingIndex]
        : taxonomy[System.Math.Min(existingIndex, nonfiniteIndex)];
}

static bool CompareReplay(ReplayAuditRow[] actualRows, ReplayAggregate actualAggregate,
    ReplayAuditRow[] reportedRows, ReplayAggregate reportedAggregate, double tolerance)
{
    if (actualRows.Length != 6 || reportedRows.Length != 6
        || actualRows.Select(row => row.ChainId).Distinct(StringComparer.Ordinal).Count() != 6
        || reportedRows.Select(row => row.ChainId).Distinct(StringComparer.Ordinal).Count() != 6)
        return false;
    foreach (ReplayAuditRow actual in actualRows)
    {
        ReplayAuditRow[] matches = reportedRows.Where(row => row.ChainId == actual.ChainId).ToArray();
        if (matches.Length != 1) return false;
        ReplayAuditRow reported = matches[0];
        if (actual.DecisionsMatch != reported.DecisionsMatch
            || actual.FinalPositionBitIdentical != reported.FinalPositionBitIdentical
            || ScaledAbsoluteDifference(actual.WorstScaledAbsoluteDeltaHDeviation,
                reported.WorstScaledAbsoluteDeltaHDeviation) > tolerance
            || ScaledAbsoluteDifference(actual.AcceptanceRate, reported.AcceptanceRate) > tolerance
            || actual.RetainedDraws != reported.RetainedDraws || actual.Passed != reported.Passed)
            return false;
    }
    return actualAggregate == reportedAggregate;
}

static MovementObservation[] BuildMovementFixture(string[] endpoints,
    Func<string, double[]> ratios, bool zeroBaseline = false)
{
    var rows = new List<MovementObservation>();
    foreach (string endpoint in endpoints)
        for (int index = 0; index < 4; index++)
        {

            double baseline = zeroBaseline && endpoint == endpoints[0] ? 0.0 : 1.0;
            rows.Add(new MovementObservation(endpoint, "baseline", baseline, baseline, baseline, baseline));
            rows.Add(new MovementObservation(endpoint, "long",

}

static MovementClassificationResult ClassifyMovement(IEnumerable<MovementObservation> observations,
    string[] endpoints, string[] panels, string baselineArm, string comparisonArm,
    int proposalsPerArmEndpoint, double improvementThreshold, double degradationFloor,
    int improvedPanelsRequired, int qualifyingEndpointsRequired, bool prerequisitesPassed)
{
    MovementObservation[] source = observations.ToArray();
    static double PanelValue(MovementObservation row, string panel) => panel switch
    {
        "E" => row.E, "W" => row.W, "C" => row.C, "Cperp" => row.Cperp, _ => double.NaN,
    };
    var endpointRows = new List<MovementEndpointRow>();
    foreach (string endpoint in endpoints)
    {
        var ratios = new Dictionary<string, double>(StringComparer.Ordinal);
        bool complete = true;
        foreach (string panel in panels)
        {
            MovementObservation[] baseline = source.Where(row =>
                row.CheckpointId == endpoint && row.Arm == baselineArm).ToArray();
            MovementObservation[] comparison = source.Where(row =>
                row.CheckpointId == endpoint && row.Arm == comparisonArm).ToArray();
            complete &= baseline.Length == proposalsPerArmEndpoint
                && comparison.Length == proposalsPerArmEndpoint;
            double baselineMean = baseline.Length == 0 ? double.NaN
                : baseline.Average(row => PanelValue(row, panel));
            double comparisonMean = comparison.Length == 0 ? double.NaN
                : comparison.Average(row => PanelValue(row, panel));
            ratios[panel] = baselineMean > 0.0 && double.IsFinite(baselineMean)
                && double.IsFinite(comparisonMean) ? comparisonMean / baselineMean : double.NaN;
        }
        bool conclusive = complete && ratios.Count == 4 && ratios.Values.All(double.IsFinite);
        int improved = conclusive ? ratios.Values.Count(value => value >= improvementThreshold) : 0;
        bool noneBelowFloor = conclusive && ratios.Values.All(value => value >= degradationFloor);
        bool qualifies = conclusive && improved >= improvedPanelsRequired && noneBelowFloor;
        endpointRows.Add(new MovementEndpointRow(endpoint, ratios["E"], ratios["W"], ratios["C"],
            ratios["Cperp"], complete, conclusive, improved, noneBelowFloor, qualifies));
    }
    double[] allRatios = endpointRows.SelectMany(row =>
        new[] { row.ERatio, row.WRatio, row.CRatio, row.CperpRatio }).ToArray();
    bool conclusiveAll = endpointRows.Count == endpoints.Length
        && endpointRows.All(row => row.Conclusive) && allRatios.All(double.IsFinite);
    int qualifyingCount = endpointRows.Count(row => row.Qualifies);
    bool supported = prerequisitesPassed && conclusiveAll
        && qualifyingCount >= qualifyingEndpointsRequired;
    bool mixed = prerequisitesPassed && conclusiveAll && !supported
        && ((allRatios.Any(value => value >= improvementThreshold)
                && allRatios.Any(value => value < improvementThreshold))
            || (allRatios.Any(value => value < degradationFloor)
                && allRatios.Any(value => value >= degradationFloor)));
    return new MovementClassificationResult([.. endpointRows], conclusiveAll,
        qualifyingCount, supported, mixed);
}

static Subspaces BuildSubspaces(SimplicialMesh mesh, int dimG, int extent)
{
    int edgeCount = mesh.EdgeCount;
    int dof = edgeCount * dimG;
    var exactCandidates = new List<Direction>();
    for (int vertex = 0; vertex < mesh.VertexCount; vertex++)
        for (int algebra = 0; algebra < dimG; algebra++)
        {
            var vector = new double[dof];
            for (int edge = 0; edge < edgeCount; edge++)
                vector[edge * dimG + algebra] =
                    (mesh.Edges[edge][1] == vertex ? 1.0 : 0.0)
                    - (mesh.Edges[edge][0] == vertex ? 1.0 : 0.0);
            exactCandidates.Add(new Direction($"E-v{vertex:D3}-a{algebra}", "exact", vector));
        }
    Direction[] exact = OrthonormalizeDirections(exactCandidates, []);

    var windingCandidates = new List<Direction>();
    for (int axis = 0; axis < 4; axis++)
        for (int algebra = 0; algebra < dimG; algebra++)
        {
            var vector = new double[dof];
            for (int edge = 0; edge < edgeCount; edge++)
            {
                ReadOnlySpan<double> c0 = mesh.GetVertexCoordinates(mesh.Edges[edge][0]);
                ReadOnlySpan<double> c1 = mesh.GetVertexCoordinates(mesh.Edges[edge][1]);
                int difference = (int)System.Math.Round(c1[axis] - c0[axis]);
                int wrapped = ((difference % extent) + extent) % extent;
                vector[edge * dimG + algebra] = wrapped == extent - 1 ? -1.0 : wrapped;
            }
            windingCandidates.Add(new Direction($"W-axis{axis}-a{algebra}", "winding", vector));
        }
    Direction[] winding = OrthonormalizeDirections(windingCandidates, exact);
    Direction[] closed = [.. exact, .. winding];

    var complementCandidates = new List<Direction>();
    for (int coordinate = 0; coordinate < dof && complementCandidates.Count < 64; coordinate++)
    {
        var vector = new double[dof];
        vector[coordinate] = 1.0;
        complementCandidates.Add(new Direction($"Cperp-coordinate-{coordinate:D4}", "complement-control", vector));
    }
    Direction[] complement = OrthonormalizeDirections(complementCandidates, closed).Take(12).ToArray();
    Direction[] all = [.. closed, .. complement];
    double maximumOrthogonality = 0.0;
    for (int i = 0; i < all.Length; i++)
        for (int j = 0; j < all.Length; j++)
            maximumOrthogonality = System.Math.Max(maximumOrthogonality,
                System.Math.Abs(Dot(all[i].Vector, all[j].Vector) - (i == j ? 1.0 : 0.0)));
    double maximumFaceResidual = 0.0;
    foreach (Direction direction in closed)
        for (int face = 0; face < mesh.FaceCount; face++)
            for (int algebra = 0; algebra < dimG; algebra++)
            {
                double residual = 0.0;
                int[] edges = mesh.FaceBoundaryEdges[face];
                int[] orientations = mesh.FaceBoundaryOrientations[face];
                for (int i = 0; i < edges.Length; i++)
                    residual += orientations[i] * direction.Vector[edges[i] * dimG + algebra];
                maximumFaceResidual = System.Math.Max(maximumFaceResidual, System.Math.Abs(residual));
            }
    return new Subspaces(exact, winding, closed, complement, all, maximumOrthogonality, maximumFaceResidual);
}

static Direction[] OrthonormalizeDirections(IEnumerable<Direction> candidates, IEnumerable<Direction> prefix)
{
    var basis = prefix.Select(row => row.Vector).ToList();
    var accepted = new List<Direction>();
    foreach (Direction candidateRow in candidates)
    {
        double[] candidate = (double[])candidateRow.Vector.Clone();
        for (int pass = 0; pass < 2; pass++)
            foreach (double[] existing in basis)
            {
                double coefficient = Dot(candidate, existing);
                for (int i = 0; i < candidate.Length; i++) candidate[i] -= coefficient * existing[i];
            }
        double norm = System.Math.Sqrt(NormSquared(candidate));
        if (norm <= 1e-10) continue;
        for (int i = 0; i < candidate.Length; i++) candidate[i] /= norm;
        int first = Array.FindIndex(candidate, value => System.Math.Abs(value) > 1e-14);
        if (first >= 0 && candidate[first] < 0)
            for (int i = 0; i < candidate.Length; i++) candidate[i] = -candidate[i];
        basis.Add(candidate);
        accepted.Add(candidateRow with { Vector = candidate });
    }
    return [.. accepted];
}

static ProjectionMetric ProjectMetric(double[] vector, Subspaces subspaces, int dimG)
{
    double total = NormSquared(vector);
    double eSquared = subspaces.Exact.Sum(direction =>
    {
        double coefficient = Dot(vector, direction.Vector);
        return coefficient * coefficient;
    });
    double wSquared = subspaces.Winding.Sum(direction =>
    {
        double coefficient = Dot(vector, direction.Vector);
        return coefficient * coefficient;
    });
    double cSquared = eSquared + wSquared;
    double cPerpSquared = System.Math.Max(0.0, total - cSquared);
    var gram = new double[3, 3];
    int scalarClosedDimension = subspaces.Closed.Length / dimG;
    for (int scalar = 0; scalar < scalarClosedDimension; scalar++)
    {
        var coefficients = new double[3];
        for (int algebra = 0; algebra < dimG; algebra++)
            coefficients[algebra] = Dot(vector, subspaces.Closed[scalar * dimG + algebra].Vector);
        for (int a = 0; a < dimG; a++)
            for (int b = 0; b < dimG; b++)
                gram[a, b] += coefficients[a] * coefficients[b];
    }
    double[] eigenvalues = Symmetric3EigenvaluesByJacobi(gram);
    double trace = eigenvalues.Sum();
    double alignment = trace > 0.0 ? eigenvalues[^1] / trace : 0.0;
    double rankOneDistance = cPerpSquared + eigenvalues[0] + eigenvalues[1];
    double residual = System.Math.Max(System.Math.Abs(cSquared - eSquared - wSquared),
        System.Math.Abs(total - cSquared - cPerpSquared)) / System.Math.Max(1.0, total);
    return new ProjectionMetric(total, eSquared, wSquared, cSquared, cPerpSquared,
        eigenvalues, alignment, rankOneDistance, residual);
}

static double[] Symmetric3EigenvaluesByJacobi(double[,] source)
{
    var matrix = (double[,])source.Clone();
    for (int sweep = 0; sweep < 32; sweep++)
    {
        int p = 0, q = 1;
        double largest = System.Math.Abs(matrix[0, 1]);
        if (System.Math.Abs(matrix[0, 2]) > largest) { p = 0; q = 2; largest = System.Math.Abs(matrix[0, 2]); }
        if (System.Math.Abs(matrix[1, 2]) > largest) { p = 1; q = 2; largest = System.Math.Abs(matrix[1, 2]); }
        if (largest <= 1e-15 * System.Math.Max(1.0,
            System.Math.Abs(matrix[0, 0]) + System.Math.Abs(matrix[1, 1]) + System.Math.Abs(matrix[2, 2]))) break;
        double angle = 0.5 * System.Math.Atan2(2.0 * matrix[p, q], matrix[q, q] - matrix[p, p]);
        double cosine = System.Math.Cos(angle), sine = System.Math.Sin(angle);
        double app = matrix[p, p], aqq = matrix[q, q], apq = matrix[p, q];
        matrix[p, p] = cosine * cosine * app - 2.0 * sine * cosine * apq + sine * sine * aqq;
        matrix[q, q] = sine * sine * app + 2.0 * sine * cosine * apq + cosine * cosine * aqq;
        matrix[p, q] = matrix[q, p] = 0.0;
        for (int r = 0; r < 3; r++)
        {
            if (r == p || r == q) continue;
            double arp = matrix[r, p], arq = matrix[r, q];
            matrix[r, p] = matrix[p, r] = cosine * arp - sine * arq;
            matrix[r, q] = matrix[q, r] = sine * arp + cosine * arq;
        }
    }
    return new[] { matrix[0, 0], matrix[1, 1], matrix[2, 2] }
        .Select(value => System.Math.Max(0.0, value)).OrderBy(value => value).ToArray();
}

static bool CompareProjection(JsonElement reported, ProjectionMetric actual, double tolerance)
{
    if (reported.ValueKind != JsonValueKind.Object) return false;
    foreach ((string name, double value) in new[]
    {
        ("totalNormSquared", actual.TotalNormSquared),
        ("eNormSquared", actual.ENormSquared),
        ("wNormSquared", actual.WNormSquared),
        ("cNormSquared", actual.CNormSquared),
        ("cperpNormSquared", actual.CperpNormSquared),
        ("closedRankOneAlignment", actual.ClosedRankOneAlignment),
        ("distanceSquaredToRankOneClosedCoordinateSet", actual.DistanceSquaredToRankOneClosedCoordinateSet),
        ("decompositionRelativeResidual", actual.DecompositionRelativeResidual),
    })
    {
        double? expected = JsonDouble(reported, name);
        if (expected is null || ScaledAbsoluteDifference(value, expected.Value) > tolerance) return false;
    }
    if (!reported.TryGetProperty("closedGramEigenvalues", out JsonElement spectrum)
        || spectrum.ValueKind != JsonValueKind.Array) return false;
    double[] expectedSpectrum = spectrum.EnumerateArray().Select(row => row.GetDouble()).ToArray();
    return expectedSpectrum.Length == actual.ClosedGramEigenvalues.Length
        && expectedSpectrum.Zip(actual.ClosedGramEigenvalues)
            .All(pair => ScaledAbsoluteDifference(pair.Second, pair.First) <= tolerance);
}

static Proposal Leapfrog(double[] initialPosition, double[] initialMomentum, Evaluation initial,
    double stepSize, int steps, double divergenceThreshold, Func<double[], Evaluation> evaluate)
{
    var q = (double[])initialPosition.Clone();
    var p = (double[])initialMomentum.Clone();
    double initialHamiltonian = initial.Action + 0.5 * NormSquared(p);
    Evaluation current = initial;
    bool finite = double.IsFinite(current.Action) && current.Gradient.All(double.IsFinite);
    int forceEvaluations = 1;
    if (finite)
        for (int i = 0; i < p.Length; i++) p[i] -= 0.5 * stepSize * current.Gradient[i];
    for (int leap = 0; finite && leap < steps; leap++)
    {
        for (int i = 0; i < q.Length; i++) q[i] += stepSize * p[i];
        current = evaluate(q);
        forceEvaluations++;
        finite = double.IsFinite(current.Action) && current.Gradient.All(double.IsFinite)
            && q.All(double.IsFinite);
        if (!finite) break;
        double kick = leap + 1 == steps ? 0.5 * stepSize : stepSize;
        for (int i = 0; i < p.Length; i++) p[i] -= kick * current.Gradient[i];
        finite &= p.All(double.IsFinite);
    }
    double finalHamiltonian = finite ? current.Action + 0.5 * NormSquared(p) : double.NaN;
    double deltaH = finalHamiltonian - initialHamiltonian;
    bool divergent = !finite || !double.IsFinite(deltaH) || System.Math.Abs(deltaH) > divergenceThreshold;
    return new Proposal(q, p, current.Action, current.Gradient, deltaH, finite, divergent, forceEvaluations);
}

static ScalarProposal ScalarLeapfrog(double q0, double p0, double stepSize, int steps,
    Func<double, double> potential, Func<double, double> gradient, double divergenceThreshold)
{
    double q = q0, p = p0;
    double initialHamiltonian = potential(q) + 0.5 * p * p;
    bool finite = double.IsFinite(initialHamiltonian);
    if (finite) p -= 0.5 * stepSize * gradient(q);
    for (int leap = 0; finite && leap < steps; leap++)
    {
        q += stepSize * p;
        double g = gradient(q);
        finite = double.IsFinite(q) && double.IsFinite(g);
        if (!finite) break;
        p -= (leap + 1 == steps ? 0.5 : 1.0) * stepSize * g;
        finite &= double.IsFinite(p);
    }
    double finalHamiltonian = finite ? potential(q) + 0.5 * p * p : double.NaN;
    double deltaH = finalHamiltonian - initialHamiltonian;
    bool divergent = !finite || !double.IsFinite(deltaH) || System.Math.Abs(deltaH) > divergenceThreshold;
    return new ScalarProposal(q, p, deltaH, finite, divergent);
}

static bool Accept(double deltaH, double logUniform, bool finite, bool divergent) =>
    finite && !divergent && double.IsFinite(deltaH)
    && logUniform <= System.Math.Min(0.0, -deltaH);

static double[][] SyntheticChains(int chainCount, int drawCount, double rho, double meanStep, ulong seed)
{
    var rng = new Xoshiro256StarStar(ExpandSeed(seed));
    var chains = new double[chainCount][];
    for (int chain = 0; chain < chainCount; chain++)
    {
        chains[chain] = new double[drawCount];
        double value = NextGaussian(rng);
        for (int draw = 0; draw < drawCount; draw++)
        {
            value = rho * value + System.Math.Sqrt(1.0 - rho * rho) * NextGaussian(rng) + meanStep;
            chains[chain][draw] = value;
        }
    }
    return chains;
}

static Diagnostic Diagnose(double[][] chains)
{
    int n = chains.Min(chain => chain.Length);
    double[][] trimmed = chains.Select(chain => chain.Take(n).ToArray()).ToArray();
    double[] pooled = trimmed.SelectMany(chain => chain).ToArray();
    if (pooled.Distinct().Count() <= 1) return new Diagnostic(double.NaN, double.NaN, double.NaN);
    double[] ranked = RankNormalize(pooled);
    double median = Median(pooled);
    double[] folded = RankNormalize(pooled.Select(value => System.Math.Abs(value - median)).ToArray());
    double[][] rankedChains = Regroup(ranked, trimmed.Length, n);
    double[][] foldedChains = Regroup(folded, trimmed.Length, n);
    double[] ordered = pooled.Order().ToArray();
    double q05 = ordered[(int)System.Math.Floor(0.05 * (ordered.Length - 1))];
    double q95 = ordered[(int)System.Math.Ceiling(0.95 * (ordered.Length - 1))];
    double[][] lower = trimmed.Select(chain =>
        chain.Select(value => value <= q05 ? 1.0 : 0.0).ToArray()).ToArray();
    double[][] upper = trimmed.Select(chain =>
        chain.Select(value => value >= q95 ? 1.0 : 0.0).ToArray()).ToArray();
    return new Diagnostic(
        System.Math.Max(SplitRhat(rankedChains), SplitRhat(foldedChains)),
        EffectiveSampleSize(Split(rankedChains)),
        System.Math.Min(EffectiveSampleSize(Split(lower)), EffectiveSampleSize(Split(upper))));
}

static double SplitRhat(double[][] chains)
{
    double[][] split = Split(chains);
    int m = split.Length, n = split.Min(chain => chain.Length);
    double[] means = split.Select(chain => chain.Take(n).Average()).ToArray();
    double within = split.Select(chain => Variance(chain.Take(n).ToArray())).Average();
    if (!(within > 0.0)) return double.NaN;
    double grand = means.Average();
    double between = n * means.Sum(mean => (mean - grand) * (mean - grand)) / (m - 1);
    return System.Math.Sqrt((((n - 1.0) / n) * within + between / n) / within);
}

static double EffectiveSampleSize(double[][] chains)
{
    int m = chains.Length, n = chains.Min(chain => chain.Length);
    double[][] trimmed = chains.Select(chain => chain.Take(n).ToArray()).ToArray();
    double[] means = trimmed.Select(chain => chain.Average()).ToArray();
    double within = trimmed.Select(Variance).Average();
    if (!(within > 0.0)) return double.NaN;
    double grand = means.Average();
    double between = n * means.Sum(mean => (mean - grand) * (mean - grand)) / (m - 1);
    double varPlus = ((n - 1.0) / n) * within + between / n;
    var rho = new double[n];
    rho[0] = 1.0;
    for (int lag = 1; lag < n; lag++)
    {
        double covariance = 0.0;
        for (int chain = 0; chain < m; chain++)
        {
            double sum = 0.0;
            for (int i = 0; i + lag < n; i++)
                sum += (trimmed[chain][i] - means[chain]) * (trimmed[chain][i + lag] - means[chain]);
            covariance += sum / n;
        }
        rho[lag] = 1.0 - (within - covariance / m) / varPlus;
    }
    double tau = -1.0, previousPair = double.PositiveInfinity;
    for (int pairIndex = 0; 2 * pairIndex + 1 < n; pairIndex++)
    {
        double pair = rho[2 * pairIndex] + rho[2 * pairIndex + 1];
        if (pair < 0.0) break;
        pair = System.Math.Min(pair, previousPair);
        previousPair = pair;
        tau += 2.0 * pair;
    }
    return tau > 0.0 ? m * n / tau : double.NaN;
}

static double MaximumHalfWindowZ(double[][] chains)
{
    double maximum = 0.0;
    foreach (double[] chain in chains)
    {
        int half = chain.Length / 2;
        double[] first = chain.Take(half).ToArray();
        double[] second = chain.Skip(chain.Length - half).ToArray();
        double standardError = System.Math.Sqrt(Variance(first) / half + Variance(second) / half);
        if (standardError > 0.0)
            maximum = System.Math.Max(maximum, System.Math.Abs((second.Average() - first.Average()) / standardError));
    }
    return maximum;
}

static double LagOneCorrelation(double[] values)
{
    if (values.Length < 3) return double.NaN;
    double mean = values.Average();
    double numerator = 0.0, denominator = 0.0;
    for (int i = 0; i < values.Length; i++) denominator += (values[i] - mean) * (values[i] - mean);
    for (int i = 0; i + 1 < values.Length; i++) numerator += (values[i] - mean) * (values[i + 1] - mean);
    return denominator > 0.0 ? numerator / denominator : double.NaN;
}

static DriftMetric ComputeDrift(double[] values, int batchCount, double threshold)
{
    int batchSize = values.Length / batchCount;
    double[] means = Enumerable.Range(0, batchCount)
        .Select(batch => values.Skip(batch * batchSize).Take(batchSize).Average()).ToArray();
    int half = batchCount / 2;
    double firstMean = means.Take(half).Average(), secondMean = means.Skip(half).Average();
    double standardError = System.Math.Sqrt(Variance(means.Take(half).ToArray()) / half
        + Variance(means.Skip(half).ToArray()) / half);
    double halfWindowZ = standardError > 0.0 ? (secondMean - firstMean) / standardError : double.NaN;
    double meanX = (batchCount - 1) / 2.0, meanY = means.Average();
    double xy = 0.0, xx = 0.0;
    for (int index = 0; index < batchCount; index++)
    {
        xy += (index - meanX) * (means[index] - meanY);
        xx += (index - meanX) * (index - meanX);
    }
    double slope = xy / xx;
    double residual = 0.0;
    for (int index = 0; index < batchCount; index++)
    {
        double predicted = meanY + slope * (index - meanX);
        residual += (means[index] - predicted) * (means[index] - predicted);
    }
    double slopeStandardError = System.Math.Sqrt(residual / (batchCount - 2) / xx);
    double slopeZ = slopeStandardError > 0.0 ? slope / slopeStandardError : double.NaN;
    bool conclusive = double.IsFinite(halfWindowZ) && double.IsFinite(slopeZ);
    return new DriftMetric(Reportable(halfWindowZ), Reportable(slopeZ), conclusive,
        conclusive && (System.Math.Abs(halfWindowZ) > threshold || System.Math.Abs(slopeZ) > threshold));
}

static double[] RankNormalize(double[] values)
{
    int[] order = Enumerable.Range(0, values.Length).OrderBy(index => values[index]).ToArray();
    var ranks = new double[values.Length];
    int cursor = 0;
    while (cursor < order.Length)
    {
        int last = cursor;
        while (last + 1 < order.Length && values[order[last + 1]] == values[order[cursor]]) last++;
        double averageRank = (cursor + last) / 2.0 + 1.0;
        for (int i = cursor; i <= last; i++) ranks[order[i]] = averageRank;
        cursor = last + 1;
    }
    return ranks.Select(rank => InverseNormalCdf((rank - 0.375) / (values.Length + 0.25))).ToArray();
}

static double InverseNormalCdf(double p)
{
    double[] a = [-39.69683028665376, 220.9460984245205, -275.9285104469687, 138.3577518672690, -30.66479806614716, 2.506628277459239];
    double[] b = [-54.47609879822406, 161.5858368580409, -155.6989798598866, 66.80131188771972, -13.28068155288572];
    double[] c = [-0.007784894002430293, -0.3223964580411365, -2.400758277161838, -2.549732539343734, 4.374664141464968, 2.938163982698783];
    double[] d = [0.007784695709041462, 0.3224671290700398, 2.445134137142996, 3.754408661907416];
    const double boundary = 0.02425;
    if (p < boundary)
    {
        double q = System.Math.Sqrt(-2.0 * System.Math.Log(p));
        return (((((c[0] * q + c[1]) * q + c[2]) * q + c[3]) * q + c[4]) * q + c[5])
            / ((((d[0] * q + d[1]) * q + d[2]) * q + d[3]) * q + 1.0);
    }
    if (p > 1.0 - boundary)
    {
        double q = System.Math.Sqrt(-2.0 * System.Math.Log(1.0 - p));
        return -(((((c[0] * q + c[1]) * q + c[2]) * q + c[3]) * q + c[4]) * q + c[5])
            / ((((d[0] * q + d[1]) * q + d[2]) * q + d[3]) * q + 1.0);
    }
    double r = p - 0.5, s = r * r;
    return (((((a[0] * s + a[1]) * s + a[2]) * s + a[3]) * s + a[4]) * s + a[5]) * r
        / (((((b[0] * s + b[1]) * s + b[2]) * s + b[3]) * s + b[4]) * s + 1.0);
}

static double[][] Orthonormalize(double[][] candidates)
{
    var basis = new List<double[]>();
    foreach (double[] source in candidates)
    {
        double[] candidate = (double[])source.Clone();
        foreach (double[] existing in basis)
        {
            double coefficient = Dot(candidate, existing);
            for (int i = 0; i < candidate.Length; i++) candidate[i] -= coefficient * existing[i];
        }
        double norm = System.Math.Sqrt(NormSquared(candidate));
        if (norm <= 1e-12) continue;
        for (int i = 0; i < candidate.Length; i++) candidate[i] /= norm;
        basis.Add(candidate);
    }
    return [.. basis];
}

static double[] Project(double[] vector, double[][] basis)
{
    var projected = new double[vector.Length];
    foreach (double[] direction in basis)
    {
        double coefficient = Dot(vector, direction);
        for (int i = 0; i < vector.Length; i++) projected[i] += coefficient * direction[i];
    }
    return projected;
}

static void FindNonfinitePaths(object? value, string path, List<string> paths, HashSet<object> active)
{
    if (value is null or string or JsonElement or JsonNode) return;
    if (value is double doubleValue)
    {
        if (!double.IsFinite(doubleValue)) paths.Add(path);
        return;
    }
    if (value is float floatValue)
    {
        if (!float.IsFinite(floatValue)) paths.Add(path);
        return;
    }
    Type type = value.GetType();
    if (type.IsPrimitive || type.IsEnum || value is decimal) return;
    if (!type.IsValueType && !active.Add(value)) return;
    try
    {
        if (value is System.Collections.IDictionary dictionary)
        {
            foreach (System.Collections.DictionaryEntry entry in dictionary)
                FindNonfinitePaths(entry.Value, path + "." + entry.Key, paths, active);
            return;
        }
        if (value is System.Collections.IEnumerable enumerable)
        {
            int index = 0;
            foreach (object? item in enumerable)
                FindNonfinitePaths(item, path + "[" + index++ + "]", paths, active);
            return;
        }
        foreach (System.Reflection.PropertyInfo property in type.GetProperties(
            System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public))
        {
            if (!property.CanRead || property.GetIndexParameters().Length != 0) continue;
            string name = JsonNamingPolicy.CamelCase.ConvertName(property.Name);
            FindNonfinitePaths(property.GetValue(value), path + "." + name, paths, active);
        }
    }
    finally
    {
        if (!type.IsValueType) active.Remove(value);
    }
}

{

}
static ulong[] ExpandSeed(ulong seed)
{
    ulong state = seed;
    ulong NextSplitMix()
    {
        state += 0x9E3779B97F4A7C15UL;
        ulong value = state;
        value = (value ^ (value >> 30)) * 0xBF58476D1CE4E5B9UL;
        value = (value ^ (value >> 27)) * 0x94D049BB133111EBUL;
        return value ^ (value >> 31);
    }
    return [NextSplitMix(), NextSplitMix(), NextSplitMix(), NextSplitMix()];
}

sealed class Xoshiro256StarStar(ulong[] state)
{
    private ulong _s0 = state[0], _s1 = state[1], _s2 = state[2], _s3 = state[3];
    public ulong NextUInt64()
    {
        ulong result = RotateLeft(_s1 * 5UL, 7) * 9UL;
        ulong t = _s1 << 17;
        _s2 ^= _s0;
        _s3 ^= _s1;
        _s1 ^= _s2;
        _s0 ^= _s3;
        _s2 ^= t;
        _s3 = RotateLeft(_s3, 45);
        return result;
    }
    private static ulong RotateLeft(ulong value, int count) => (value << count) | (value >> (64 - count));
}

{
    public override double Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options) =>
        throw new NotSupportedException("Phase572 serialization converter is write-only.");

    {
        if (double.IsFinite(value)) writer.WriteNumberValue(value);

    }
